'use strict';

/**
 * Knowledge chunking with overlapping windows and paragraph-aware splits.
 *
 * Strategy:
 *  - Split the raw text into "paragraphs" (double newlines separate; if the
 *    doc has no blank lines, fall back to whole-text windows).
 *  - Oversized paragraphs (longer than `window`) are split into overlapping
 *    chunks with `overlap` characters of context carried over.
 *  - Small paragraphs are packed together up to `window` size so chunks
 *    remain context-rich instead of tiny.
 *
 * @param {string} text
 * @param {{window?: number, overlap?: number}} [opts]
 * @returns {Array<{index:number, text:string}>}
 */
function chunkText(text, opts = {}) {
  const window = opts.window || 400;
  const overlap = Math.min(opts.overlap ?? 40, window - 1);
  const src = String(text == null ? '' : text).trim();
  if (!src) return [];

  const chunks = [];

  // Split an over-long run, carrying `overlap` chars between windows.
  const emitRun = (run) => {
    let rest = run;
    while (rest.length > window) {
      let cut = window;
      // prefer cutting near a sentence boundary inside the first window
      const breaks = rest.slice(0, window).search(/[。！？!?；;]/);
      if (breaks !== -1 && breaks > window * 0.5) cut = breaks + 1;
      chunks.push(rest.slice(0, cut).trim());
      rest = rest.slice(cut - overlap);
    }
    const tail = rest.trim();
    if (tail) chunks.push(tail);
  };

  // Split text into paragraph-ish units, then pack/oversize each.
  // Splitting on single newlines keeps markdown QA blocks and numbered list
  // items as distinct paragraphs; the pack-down step re-groups related lines.
  const paras = src
    .split(/\n/)
    .map((p) => p.trim())
    .filter((p) => p.length > 0);

  if (paras.length === 1 && paras[0].length === src.length && !src.includes('\n')) {

    }
  }
  if (buf.trim()) chunks.push(buf.trim());

  return chunks
    .filter((c) => c.length > 0)
    .map((c, i) => ({ index: i + 1, text: c }));
}

/**
 * Chinese-friendly FTS5 tokenization utilities.
 *
 * SQLite FTS5 does not segment Chinese text out of the box. We use a
 * "unicode61 + separators" scheme: characters are split on punctuation,
 * whitespace and Latin/CJK boundary crossings, while runs of Han characters
 * are additionally n-grammed so short queries (2~3 chars) still hit the
 * index. Benign ascii words (english / numbers / urls) are kept whole.
 */

/**
 * Split text into tokens, classifying each run as 'cjk' (Han) or 'other'.
 * Punctuation and whitespace break tokens and are dropped.
 */
function tokenize(text) {
  const t = String(text == null ? '' : text);
  const out = [];
  let buf = '';
  let bufIsCJK = null;

  const flush = () => {
    if (buf) out.push({ kind: bufIsCJK ? 'cjk' : 'other', token: buf });
    buf = '';
    bufIsCJK = null;
  };

  for (const ch of t) {
    if (DELIM_RE.test(ch)) {
      flush();
      continue;
    }
    const isCJK = CJK_RE.test(ch);
    if (bufIsCJK === null) bufIsCJK = isCJK;
    else if (isCJK !== bufIsCJK) {
      flush();
      bufIsCJK = isCJK;
    }
    buf += ch;
  }
  flush();
  return out;
}

/** Tokenize a CJK run into overlapping n-grams (default bigrams). */
function ngrams(cjkRun, n = 2) {
  const out = [];
  if (!cjkRun) return out;
  if (cjkRun.length < n) {
    out.push(cjkRun);
    return out;
  }
  for (let i = 0; i <= cjkRun.length - n; i++) out.push(cjkRun.slice(i, i + n));
  return out;
}

 * A pure-Chinese run becomes an OR-group of its bigrams, e.g.
 *   律师咨询费怎么收取  ->  ("律师" OR "师咨" OR "咨询" OR "询费" OR
 *                          "费怎" OR "怎么" OR "么收" OR "收取")
 * so a chunk that shares even a few bigrams is retrieved and BM25 ranks by
 * overlap. Latin/number runs keep AND semantics ("400" "123" "4567").
 *
 * Returns '' when no usable terms (caller falls back to a LIKE scan).
 */
function toQuery(text) {
  const groups = [];
  for (const { kind, token } of tokenize(normalize(text))) {
    if (kind === 'cjk') {
      if (token.length === 1) groups.push(`"${token}"`);
      else groups.push(`(${ngrams(token, 2).map((g) => `"${g}"`).join(' OR ')})`);
    } else {
      groups.push(`"${token}"`);
    }
  }
  return groups.join(' ');
}

/** Build the expansion list for indexing a chunk. */
function expandForIndex(text) {
  return tokenize(text).flatMap(({ kind, token }) =>
    kind === 'cjk' ? ngrams(token, 2) : [normalize(token)]
  );
}

/** Strip the trailing "-- N of M --" page markers pdf-parse appends. */
function stripPageMarkers(text) {
  return String(text).replace(/--\s*\d+\s+of\s+\d+\s*--/g, '').trim();
}

/** PDF via pdf-parse, stripping trailing page-marker lines. */
async function extractPdf(buffer) {
  const { stripPageMarkers } = require('./tokenizer');
  const PDFParse = require('pdf-parse').PDFParse;
  const parser = new PDFParse({ data: new Uint8Array(buffer) });
  const out = await parser.getText();
  let text = Array.isArray(out) ? out.join('') : out.text || '';
  text = text.replace(/[\u0000\uFFFD]+/g, ' '); // strip glyph holes
  return stripPageMarkers(text);
}

/** Extract plain text from any format we know, by extension. */
function textFromBuffer(filename, buffer) {
  const ext = path.extname(filename).toLowerCase();
  switch (ext) {
    case '.md':
    case '.markdown':
    case '.txt':
      return buffer.toString('utf8');
    case '.csv':
      return parseCsv(buffer.toString('utf8'))
        .map((r) => r.join('\t'))
        .join('\n');
    default:
      return null;
  }
}

/**
 * Coerce a user-supplied limit into a sane integer. Malformed values (e.g.
 * "abc") would otherwise pass NaN into SQLite LIMIT ? and blow up with a
 * "datatype mismatch" 500; negative/zero/huge values are clamped too.
 * @param {*} v raw value from query/body
 * @param {number} dflt fallback when v is not a positive finite number
 * @returns {number}
 */
function safeLimit(v, dflt) {
  const n = Math.floor(Number(v));
  return Number.isFinite(n) && n >= 1 ? Math.min(n, 1000) : dflt;
}

          : err && err.type === 'entity.too.large'
            ? 'request body too large'
            : err && err.message
              ? err.message
              : 'internal server error';
    if (status >= 500) console.error('[api]', err);
    res.status(status).json({ error: message });
  });

  return app;
}

module.exports = { buildApp };
